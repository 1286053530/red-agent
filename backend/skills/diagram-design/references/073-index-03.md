    <script>
      const state = { type: "architecture", variant: "" };
      const iframe = document.getElementById("preview");
      const openRaw = document.getElementById("open-raw");

      const nav = document.getElementById("diagram-nav");
      const navToggle = document.getElementById("nav-toggle");
      const search = document.getElementById("diagram-search");
      const typeTabs = [...document.querySelectorAll("#type-tabs .tab")];
      const diagramName = (tab) => [...tab.childNodes]
        .filter((node) => node.nodeType === Node.TEXT_NODE)
        .map((node) => node.textContent).join("").trim();

      function setNavigation(open) {
        nav.hidden = !open;
        navToggle.setAttribute("aria-expanded", String(open));
      }
      navToggle.addEventListener("click", () => setNavigation(nav.hidden));
      nav.addEventListener("keydown", (event) => {
        if (event.key === "Escape") {
          setNavigation(false);
          navToggle.focus();
        }
      });
      function filterDiagrams() {
        const query = search.value.trim().toLowerCase();
        typeTabs.forEach((tab) => {
          tab.hidden = !diagramName(tab).toLowerCase().includes(query);
        });
        const visible = typeTabs.filter((tab) => !tab.hidden);
        // Keep one visible option reachable with Tab even if selection is filtered out.
        const focusable = visible.find((tab) => tab.dataset.type === state.type) || visible[0];
        typeTabs.forEach((tab) => { tab.tabIndex = tab === focusable ? 0 : -1; });
        document.getElementById("empty-results").hidden = visible.length > 0;
      }
      search.addEventListener("input", filterDiagrams);
      // Collapse on click, not pointerdown: hiding the catalog moves the preview
      // up, so collapsing mid-press would pull a control out from under the pointer
      // and its click would never fire.
      document.querySelector("main").addEventListener("click", () => setNavigation(false));
      iframe.addEventListener("load", () => {
        try {
          // Iframe events do not bubble to the gallery. Bind after each example load.
          iframe.contentDocument.addEventListener("click", () => setNavigation(false));
        } catch {
          // Local file previews may have an opaque origin; focus is handled below.
        }
      });
      // Opaque local previews expose only the focus change on press. Wait for a
      // normal click to finish inside the example before moving it.
      const PREVIEW_CLICK_GRACE_MS = 400;
      let blurCollapse = 0;
      window.addEventListener("blur", () => {
        if (document.activeElement !== iframe) return;
        let accessible = false;
        try { accessible = iframe.contentDocument !== null; } catch { accessible = false; }
        if (accessible) return;
        clearTimeout(blurCollapse);
        blurCollapse = setTimeout(() => {
          if (document.activeElement === iframe) setNavigation(false);
        }, PREVIEW_CLICK_GRACE_MS);
      });

      function update() {
        const activeType = [...document.querySelectorAll("#type-tabs .tab")].find(
          (tab) => tab.dataset.type === state.type,
        );
        const name = diagramName(activeType);
        document.getElementById("current-diagram").textContent = name;
        iframe.title = `${name} preview`;
        const single = activeType && activeType.hasAttribute("data-single");
        const variantTabs = document.querySelectorAll("#variant-tabs .tab");
        if (single) state.variant = "";
        variantTabs.forEach((tab) => {
          tab.disabled = single && tab.dataset.variant !== "";
        });
        for (const [containerId, key] of [
          ["type-tabs", "type"],
          ["variant-tabs", "variant"],
        ]) {
          document.querySelectorAll(`#${containerId} .tab`).forEach((tab) => {
            const active = tab.dataset[key] === state[key];
            tab.classList.toggle("active", active);
            tab.setAttribute("aria-checked", String(active));
            tab.tabIndex = active ? 0 : -1;
          });
        }
        const src = `example-${state.type}${state.variant}.html`;
        iframe.src = src;
        openRaw.href = src;
        filterDiagrams();
        const selectedTabs = document.querySelectorAll('#variant-tabs [role="radio"][aria-checked="true"]');
        iframe.setAttribute(
          "aria-labelledby",
          ["current-diagram", ...[...selectedTabs].map((tab) => tab.id)].join(" "),
        );
      }

      function bindTabs(containerId, key) {
        const container = document.getElementById(containerId);
        const tabs = [...container.querySelectorAll(".tab")];
        tabs.forEach((tab, index) => {
          tab.id = `${containerId}-${index}`;
          tab.setAttribute("role", "radio");
          tab.setAttribute("aria-controls", "preview");
        });

        function activate(tab, focus = false) {
          state[key] = tab.dataset[key];
          update();
          if (focus) tab.focus();
        }

        container.addEventListener("click", (e) => {
          const btn = e.target.closest(".tab");
          if (!btn || btn.disabled) return;
          activate(btn);
        });
        container.addEventListener("keydown", (e) => {
          const current = e.target.closest('[role="radio"]');
          if (!current || !container.contains(current)) return;
          const enabledTabs = tabs.filter((tab) => !tab.disabled && !tab.hidden);
          const index = enabledTabs.indexOf(current);
          let next;
          if (e.key === "ArrowRight" || e.key === "ArrowDown") {
            next = enabledTabs[(index + 1) % enabledTabs.length];
          }
          if (e.key === "ArrowLeft" || e.key === "ArrowUp") {
            next = enabledTabs[(index - 1 + enabledTabs.length) % enabledTabs.length];
          }
          if (e.key === "Home") next = enabledTabs[0];
          if (e.key === "End") next = enabledTabs.at(-1);
          if (!next) return;
          e.preventDefault();
          activate(next, true);
        });
      }

      bindTabs("type-tabs", "type");
      bindTabs("variant-tabs", "variant");
      update();
    </script>
